  <meta charset="UTF-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no" />
  <title>Kanizsa Illusion Puzzle</title>
  <script src="https://cdn.tailwindcss.com"></script>
  <script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
  <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
  <style>
    body { margin: 0; overflow: hidden; background-color: #ffffff; touch-action: none; }
    .fade-transition { transition: opacity 0.5s ease-in-out; }
    .v-transition { transition: transform 1.2s ease-in-out; }
  </style>
</head>
<body>
  <div id="root"></div>

  <script type="text/babel">
    const { useState, useEffect, useRef } = React;

    // === 絶対的な幾何学定数 ===
    const R = 150; // 外側の基準半径
    const r = 30;  // パックマンの半径
    const D_DIAMOND = 30 * Math.sqrt(3); // 51.9615 (ダイヤ形成時の中心距離)
    const E_THIRD = R * Math.sqrt(3) / 3; // 86.6025 (V字の線の長さ)

    // === 各フェーズのターゲット座標 ===
    const P1_TARGETS = [
      { x: 0, y: -D_DIAMOND }, // 上
      { x: D_DIAMOND * 0.866025, y: D_DIAMOND * 0.5 }, // 右下
      { x: -D_DIAMOND * 0.866025, y: D_DIAMOND * 0.5 } // 左下
    ];

    const P2_TARGETS = [
      { x: 0, y: -R },
      { x: R * 0.866025, y: R * 0.5 },
      { x: -R * 0.866025, y: R * 0.5 }
    ];

    const V1_POSITIONS = [
      { x: 0, y: 0 }, { x: 0, y: 0 }, { x: 0, y: 0 }
    ];

    const V2_POSITIONS = [
      { x: 0, y: R }, // 下
      { x: -R * 0.866025, y: -R * 0.5 }, // 左上
      { x: R * 0.866025, y: -R * 0.5 } // 右上
    ];

    const W_TARGETS = [
      { x: 0, y: -R - 30 },
      { x: (R + 30) * 0.866025, y: (R + 30) * 0.5 },
      { x: -(R + 30) * 0.866025, y: (R + 30) * 0.5 }
    ];

    // SVGパス: 完全に60度開いたパックマン (右向き)
    const PACMAN_PATH = `M 0 0 L 25.9808 15 A 30 30 0 1 0 25.9808 -15 Z`;
    
    // V字のSVGパス (中心から外側へ伸びる)
    const V_PATHS = [
      `M -43.3013 -75 L 0 0 L 43.3013 -75`, // V0: 上へ開く
      `M 86.6025 0 L 0 0 L 43.3013 75`,     // V1: 右下へ開く
      `M -43.3013 75 L 0 0 L -86.6025 0`    // V2: 左下へ開く
    ];

    function App() {
      const svgRef = useRef(null);
      const [phase, setPhase] = useState(1);
      const [pPos, setPPos] = useState([
        { x: P1_TARGETS[0].x, y: P1_TARGETS[0].y },
        { x: -100, y: 160 },
        { x: 100, y: 160 }
      ]);
      const [vPos, setVPos] = useState(V1_POSITIONS);
      const [wPos, setWPos] = useState({ x: 0, y: 0 });
      const [clearText, setClearText] = useState(false);
      
      const [dragging, setDragging] = useState(null);

      const isSnapped = (pos, target) => {
        return Math.hypot(pos.x - target.x, pos.y - target.y) < 40;
      };

      const handlePointerDown = (e, type, index = null) => {
        if ((phase === 1 || phase === 2) && type === 'p') {
          setDragging({ type, index, startX: e.clientX, startY: e.clientY, initialX: pPos[index].x, initialY: pPos[index].y });
        } else if (phase === 3 && type === 'w') {
          setDragging({ type, startX: e.clientX, startY: e.clientY, initialX: wPos.x, initialY: wPos.y });
        }
      };

      const handlePointerMove = (e) => {
        if (!dragging || !svgRef.current) return;
        const svg = svgRef.current;
        const pt = svg.createSVGPoint();
        pt.x = e.clientX; pt.y = e.clientY;
        const startPt = svg.createSVGPoint();
        startPt.x = dragging.startX; startPt.y = dragging.startY;
        
        const CTM = svg.getScreenCTM().inverse();
        const currentSvgPt = pt.matrixTransform(CTM);
        const startSvgPt = startPt.matrixTransform(CTM);
        
        const dx = currentSvgPt.x - startSvgPt.x;
        const dy = currentSvgPt.y - startSvgPt.y;

        if (dragging.type === 'p') {
          const newPos = [...pPos];
          newPos[dragging.index] = { x: dragging.initialX + dx, y: dragging.initialY + dy };
          setPPos(newPos);
        } else if (dragging.type === 'w') {
          setWPos({ x: dragging.initialX + dx, y: dragging.initialY + dy });
        }
      };

      const handlePointerUp = () => {
        if (!dragging) return;
        
        if (dragging.type === 'p') {
          const newPos = [...pPos];
          const current = newPos[dragging.index];
          const targets = phase === 1 ? P1_TARGETS : P2_TARGETS;
          
          if (isSnapped(current, targets[dragging.index])) {
            newPos[dragging.index] = { ...targets[dragging.index] };
          }
          setPPos(newPos);
          
          // Phase 1 -> 1.5 (V Move Animation)
          if (phase === 1 && newPos.every((p, i) => Math.hypot(p.x - P1_TARGETS[i].x, p.y - P1_TARGETS[i].y) < 1)) {
            setPhase(1.5);
            setVPos(V2_TARGETS);
            setTimeout(() => setPhase(2), 1200);
          }
          
          // Phase 2 -> 2.5 -> 3 (Kanizsa Formation)
          if (phase === 2 && newPos.every((p, i) => Math.hypot(p.x - P2_TARGETS[i].x, p.y - P2_TARGETS[i].y) < 1)) {
            setPhase(2.5);
            setTimeout(() => setPhase(3), 1200);
          }
        } else if (dragging.type === 'w') {
          let snapped = false;
          let finalPos = { ...wPos };
          for (let target of W_TARGETS) {
            if (isSnapped(wPos, target)) {
              finalPos = { ...target };
              snapped = true;
              break;
            }
          }
          setWPos(finalPos);
          
          if (snapped) {
            setPhase(4);
            setClearText(true);
            setTimeout(() => setClearText(false), 2000);
          }
        }
        setDragging(null);
      };

      useEffect(() => {
        window.addEventListener('pointermove', handlePointerMove);
        window.addEventListener('pointerup', handlePointerUp);
        return () => {
          window.removeEventListener('pointermove', handlePointerMove);
          window.removeEventListener('pointerup', handlePointerUp);
        };
      }, [dragging, pPos, wPos, phase]);

      const resetGame = () => {
        setPhase(1);
        setPPos([
          { x: P1_TARGETS[0].x, y: P1_TARGETS[0].y },
          { x: -100, y: 160 },
          { x: 100, y: 160 }
        ]);
        setVPos(V1_POSITIONS);
        setWPos({ x: 0, y: 0 });
        setClearText(false);
        setDragging(null);
      };

      // UI Text logic
      let uiText = "";
      if (phase === 1 || phase === 1.5) {
        const count = pPos.filter((p, i) => Math.hypot(p.x - P1_TARGETS[i].x, p.y - P1_TARGETS[i].y) < 1).length;
        uiText = "♦".repeat(count) + "♢".repeat(3 - count);
      } else if (phase === 2) {
        uiText = "△△";
      } else if (phase === 2.5) {
        uiText = "▲▲";
      } else {
        uiText = "◯◯△";
      }

      return (
        <div className="w-screen h-screen flex flex-col items-center justify-center relative overflow-hidden bg-white select-none">
          
          {/* Header UI */}
          <div className="absolute top-10 flex flex-col items-center z-10">
            <div className="backdrop-blur-md bg-white/50 px-8 py-4 rounded-2xl shadow-[0_4px_30px_rgba(0,0,0,0.1)] border border-white/40 text-4xl tracking-[0.3em] font-light text-gray-800">
              {uiText}
            </div>
          </div>

          {/* Clear Text */}
          <div className={`absolute inset-0 flex items-center justify-center z-20 pointer-events-none fade-transition ${clearText ? 'opacity-100' : 'opacity-0'}`}>
            <h1 className="text-6xl font-light tracking-[0.5em] text-gray-800 ml-6">clear</h1>
          </div>

          <svg 
            ref={svgRef}
            viewBox="-250 -250 500 500" 
            className="w-full h-full max-w-3xl max-h-3xl"
          >
            {/* Background Lines (V's or Full Triangle) */}
            {phase < 3 ? (
              <g stroke="black" strokeWidth="2" fill="none">
                <path d={V_PATHS[0]} style={{ transform: `translate(${vPos[0].x}px, ${vPos[0].y}px)` }} className="v-transition" />
                <path d={V_PATHS[1]} style={{ transform: `translate(${vPos[1].x}px, ${vPos[1].y}px)` }} className="v-transition" />
                <path d={V_PATHS[2]} style={{ transform: `translate(${vPos[2].x}px, ${vPos[2].y}px)` }} className="v-transition" />
              </g>
            ) : (
              <polygon points="0,150 -129.9038,-75 129.9038,-75" stroke="black" strokeWidth="2" fill="none" />
            )}

            {/* Black shapes (Pacmen or Circles) */}
            {phase < 3 ? (
              <g fill="black">
                {pPos.map((pos, i) => (
                  <path 
                    key={`p-${i}`}
                    d={PACMAN_PATH}
                    transform={`translate(${pos.x}, ${pos.y}) rotate(${[90, 210, 330][i]})`}
                    onPointerDown={(e) => handlePointerDown(e, 'p', i)}
                    style={{ cursor: phase <= 2 ? (dragging?.index === i ? 'grabbing' : 'grab') : 'default' }}
                  />
                ))}
              </g>
            ) : (
              <g fill="black">
                {P2_TARGETS.map((pos, i) => (
                  <circle key={`c-${i}`} cx={pos.x} cy={pos.y} r={r} />
                ))}
              </g>
            )}

            {/* Draggable White Triangle (Phase 3 & 4) */}
            {phase >= 3 && (
              <polygon 
                points={`0,-150 129.9038,75 -129.9038,75`}
                fill="#ffffff"
                transform={`translate(${wPos.x}, ${wPos.y})`}
                onPointerDown={(e) => handlePointerDown(e, 'w')}
                style={{ cursor: dragging?.type === 'w' ? 'grabbing' : 'grab' }}
              />
            )}
          </svg>

          {/* Reset Button */}
          <button 
            onClick={resetGame}
            className="absolute bottom-10 px-6 py-2 rounded-full text-gray-400 border border-gray-200 hover:bg-gray-50 transition-colors z-10"
          >
            Reset
          </button>
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
</body>
